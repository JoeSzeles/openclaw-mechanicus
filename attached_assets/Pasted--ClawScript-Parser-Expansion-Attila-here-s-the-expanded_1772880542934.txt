// Token types
const TOKEN_TYPES = {
  KEYWORD: 'keyword',
  IDENTIFIER: 'identifier',
  STRING: 'string',
  NUMBER: 'number',
  OPERATOR: 'operator',
  PUNCTUATION: 'punctuation',
  COMMENT: 'comment'
};

// Lexer
function lexer(code) {
  const tokens = [];
  const regex = /\/\/.*|\/\*[\s\S]*?\*\/|"(?:\\[^]|\[^\\"])*"|\d+\.?\d*|[a-zA-Z_]\w*|[=><!&|+-/*%]+|[(){}[\];,]/g;
  let match;
  while ((match = regex.exec(code))) {
    let value = match[0];
    let type;
    if (value.startsWith('//') || value.startsWith('/*')) {
      type = TOKEN_TYPES.COMMENT;
    } else if (value.startsWith('"')) {
      type = TOKEN_TYPES.STRING;
      value = value.slice(1, -1);
    } else if (/^\d/.test(value)) {
      type = TOKEN_TYPES.NUMBER;
      value = parseFloat(value);
    } else if (/^[a-zA-Z_]/.test(value)) {
      type = TOKEN_TYPES.IDENTIFIER;
    } else if (/[=><!&|+-/*%]/.test(value)) {
      type = TOKEN_TYPES.OPERATOR;
    } else {
      type = TOKEN_TYPES.PUNCTUATION;
    }
    if (type !== TOKEN_TYPES.COMMENT) {
      tokens.push({ type, value });
    }
  }
  return tokens;
}

// Parser
class ClawScriptParser {
  constructor(tokens) {
    this.tokens = tokens;
    this.pos = 0;
    this.ast = { type: 'Program', body: [] };
    this.variables = new Map();
    this.imports = new Set(['indicators']);
  }

  current() {
    return this.tokens[this.pos];
  }

  eat(type) {
    const token = this.current();
    if (token && token.type === type) {
      this.pos++;
      return token;
    }
    throw new Error(`Parse Error: Expected ${type}, got ${token ? token.type : 'EOF'} at position ${this.pos}`);
  }

  peek() {
    return this.tokens[this.pos + 1];
  }

  parse() {
    while (this.pos < this.tokens.length) {
      const stmt = this.parseStatement();
      if (stmt) this.ast.body.push(stmt);
    }
    return this.ast;
  }

  parseStatement() {
    const token = this.current();
    if (!token) return null;

    const upperValue = token.value.toUpperCase();
    switch (upperValue) {
      case 'DEF':
      case 'SET':
        return this.parseVarDeclaration(upperValue === 'DEF');
      case 'BUY':
      case 'SELL':
      case 'SELLSHORT':
        return this.parseTradeCommand(upperValue);
      case 'EXIT':
      case 'CLOSE':
        return this.parseExitCommand(upperValue);
      case 'TRAILSTOP':
        return this.parseTrailStop();
      case 'IF':
        return this.parseIfStatement();
      case 'LOOP':
      case 'WHILE':
        return this.parseLoop(upperValue);
      case 'AI_QUERY':
        return this.parseAIQuery();
      case 'AI_GENERATE_SCRIPT':
        return this.parseAIGenerate();
      case 'ANALYZE_LOG':
        return this.parseAnalyzeLog();
      case 'RUN_ML':
        return this.parseRunML();
      case 'CLAW_WEB':
        return this.parseClawWeb();
      case 'CLAW_X':
        return this.parseClawX();
      case 'CLAW_PDF':
        return this.parseClawPdf();
      case 'CLAW_IMAGE':
        return this.parseClawImage();
      case 'CLAW_VIDEO':
        return this.parseClawVideo();
      case 'CLAW_IMAGE_VIEW':
        return this.parseClawImageView();
      case 'SPAWN_AGENT':
        return this.parseSpawnAgent();
      case 'CALL_SESSION':
        return this.parseCallSession();
      case 'MUTATE_CONFIG':
        return this.parseMutateConfig();
      case 'ALERT':
        return this.parseAlert();
      case 'STORE_VAR':
        return this.parseStoreVar();
      case 'LOAD_VAR':
        return this.parseLoadVar();
      case 'WAIT':
        return this.parseWait();
      case 'TRY':
        return this.parseTryCatch();
      case 'ERROR':
        return this.parseError();
      case 'INCLUDE':
        return this.parseInclude();
      case 'OPTIMIZE':
        return this.parseOptimize();
      case 'INDICATOR':
        return this.parseIndicator();
      default:
        if (token.type === TOKEN_TYPES.IDENTIFIER) {
          return this.parseAssignment();
        }
        throw new Error(`Unexpected token: ${token.value}`);
    }
  }

  parseVarDeclaration(isDef) {
    this.eat(TOKEN_TYPES.KEYWORD); // DEF/SET
    const name = this.eat(TOKEN_TYPES.IDENTIFIER).value;
    this.eat(TOKEN_TYPES.OPERATOR); // =
    const value = this.parseExpression();
    if (isDef && this.variables.has(name)) throw new Error(`Var ${name} already defined`);
    this.variables.set(name, value);
    return { type: 'VarDecl', name, value, isDef };
  }

  parseTradeCommand(type) {
    this.eat(TOKEN_TYPES.KEYWORD); // BUY/SELL/SELLSHORT
    let size = 'config.size';
    if (this.current().type === TOKEN_TYPES.NUMBER) {
      size = this.eat(TOKEN_TYPES.NUMBER).value;
    }
    let orderType = 'MARKET';
    let price = null;
    if (this.current().value.toUpperCase() === 'AT') {
      this.eat(TOKEN_TYPES.KEYWORD); // AT
      orderType = this.eat(TOKEN_TYPES.IDENTIFIER).value.toUpperCase();
      if (orderType === 'LIMIT') {
        this.eat(TOKEN_TYPES.PUNCTUATION); // (
        price = this.parseExpression();
        this.eat(TOKEN_TYPES.PUNCTUATION); // )
      }
    }
    let condition = true;
    if (this.current().value.toUpperCase() === 'IF') {
      this.eat(TOKEN_TYPES.KEYWORD);
      condition = this.parseExpression();
    }
    let stop = 'config.stopDistance';
    if (this.current().value.toUpperCase() === 'STOP') {
      this.eat(TOKEN_TYPES.KEYWORD);
      stop = this.parseExpression();
    }
    let limit = 'config.limitDistance';
    if (this.current().value.toUpperCase() === 'LIMIT') {
      this.eat(TOKEN_TYPES.KEYWORD);
      limit = this.parseExpression();
    }
    let reason = '"Auto"';
    if (this.current().value.toUpperCase() === 'REASON') {
      this.eat(TOKEN_TYPES.KEYWORD);
      reason = this.parseExpression();
    }
    this.imports.add('trade');
    return { type: 'Trade', command: type, size, orderType, price, condition, stop, limit, reason };
  }

  parseLoop(loopType) {
    this.eat(TOKEN_TYPES.KEYWORD); // LOOP/WHILE
    let condition = null;
    if (loopType === 'WHILE') {
      condition = this.parseExpression();
    } else {
      const num = this.parseExpression();
      this.eat(TOKEN_TYPES.KEYWORD); // TIMES
      condition = { type: 'LoopCount', num };
    }
    const body = [];
    const endKey = loopType === 'LOOP' ? 'ENDLOOP' : 'ENDWHILE';
    while (this.current() && this.current().value.toUpperCase() !== endKey) {
      body.push(this.parseStatement());
    }
    this.eat(TOKEN_TYPES.KEYWORD); // ENDLOOP/ENDWHILE
    return { type: 'Loop', loopType, condition, body };
  }

  parseTryCatch() {
    this.eat(TOKEN_TYPES.KEYWORD); // TRY
    const tryBody = [];
    while (this.current() && this.current().value.toUpperCase() !== 'CATCH') {
      tryBody.push(this.parseStatement());
    }
    this.eat(TOKEN_TYPES.KEYWORD); // CATCH
    const catchVar = this.eat(TOKEN_TYPES.IDENTIFIER).value;
    const catchBody = [];
    while (this.current() && this.current().value.toUpperCase() !== 'ENDTRY') {
      catchBody.push(this.parseStatement());
    }
    this.eat(TOKEN_TYPES.KEYWORD); // ENDTRY
    return { type: 'TryCatch', tryBody, catchVar, catchBody };
  }

  parseAIQuery() {
    this.eat(TOKEN_TYPES.KEYWORD); // AI_QUERY
    const prompt = this.eat(TOKEN_TYPES.STRING).value;
    let tool = null;
    if (this.current().value.toUpperCase() === 'TOOL') {
      this.eat(TOKEN_TYPES.KEYWORD);
      tool = this.parseExpression();
    }
    let arg = null;
    if (this.current().value.toUpperCase() === 'ARG') {
      this.eat(TOKEN_TYPES.KEYWORD);
      arg = this.parseExpression();
    }
    this.imports.add('ai');
    return { type: 'AIQuery', prompt, tool, arg };
  }

  // [Add similar parse methods for other commands, e.g., parseAlert, parseError, parseWait, parseClawWeb, etc. Stub example:]

      this.eat(TOKEN_TYPES.KEYWORD);
      level = this.parseExpression();
    }
    let to = null;
    if (this.current().value.toUpperCase() === 'TO') {
      this.eat(TOKEN_TYPES.KEYWORD);
      to = this.parseExpression();
    }
    this.imports.add('alert');
    return { type: 'Alert', message, level, to };
  }

    let js = `const BaseStrategy = require('./base-strategy.cjs');\n`;

    js += '  static get STRATEGY_TYPE() { return "custom-${strategyName.toLowerCase()}"; }\n';

  }

}

// Main function
function parseAndGenerate(code, strategyName) {
  const tokens = lexer(code);
  const parser = new ClawScriptParser(tokens);
  const ast = parser.parse();
  return parser.generateJS(ast, strategyName);
}
