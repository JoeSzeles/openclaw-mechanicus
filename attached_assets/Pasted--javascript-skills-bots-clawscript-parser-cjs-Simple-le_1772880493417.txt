```javascript
// skills/bots/clawscript-parser.cjs
// Simple lexer/parser for ClawScript DSL
// Translates ClawScript code to JS for BaseStrategy extension
// Handles trading commands, AI calls, data fetch, agent ops, etc.
// Uses regex-based tokenizing + recursive descent for simplicity
// Output: String of JS code that can be written to a .cjs file

const fs = require('fs');
const path = require('path');

// Token types
const TOKEN_TYPES = {
  KEYWORD: 'keyword',
  IDENTIFIER: 'identifier',
  STRING: 'string',
  NUMBER: 'number',
  OPERATOR: 'operator',
  PUNCTUATION: 'punctuation',
  COMMENT: 'comment'
};

// Lexer function: Tokenize input code
function lexer(code) {
  const tokens = [];
  const regex = /\/\/.*|\/\*[\s\S]*?\*\/|"(?:\\[^]|\[^\\"])*"|\d+\.?\d*|[a-zA-Z_]\w*|[=><!&|+-/*%]+|[(){}[\];,]/g;
  let match;
  while ((match = regex.exec(code))) {
    const value = match[0];
    if (value.startsWith('//') || value.startsWith('/*')) {
      tokens.push({ type: TOKEN_TYPES.COMMENT, value });
    } else if (value.startsWith('"')) {
      tokens.push({ type: TOKEN_TYPES.STRING, value: value.slice(1, -1) });
    } else if (/^\d/.test(value)) {
      tokens.push({ type: TOKEN_TYPES.NUMBER, value: parseFloat(value) });
    } else if (/^[a-zA-Z_]/.test(value)) {
      tokens.push({ type: TOKEN_TYPES.IDENTIFIER, value });
    } else if (/[=><!&|+-/*%]/.test(value)) {
      tokens.push({ type: TOKEN_TYPES.OPERATOR, value });
    } else {
      tokens.push({ type: TOKEN_TYPES.PUNCTUATION, value });
    }
  }
  return tokens.filter(t => t.type !== TOKEN_TYPES.COMMENT); // Ignore comments
}

// Parser class: Recursive descent to build AST, then generate JS
class ClawScriptParser {
  constructor(tokens) {
    this.tokens = tokens;
    this.pos = 0;
    this.ast = { type: 'Program', body: [] };
    this.variables = new Map(); // Track defined vars
    this.imports = new Set(); // Track needed OpenClaw imports
  }

  current() {
    return this.tokens[this.pos];
  }

  eat(type) {
    const token = this.current();
    if (token && token.type === type) {
      this.pos++;
      return token;
    }
    throw new Error(`Expected ${type}, got ${token ? token.type : 'EOF'}`);
  }

  parse() {
    while (this.pos < this.tokens.length) {
      const stmt = this.parseStatement();
      if (stmt) this.ast.body.push(stmt);
    }
    return this.ast;
  }

  parseStatement() {
    const token = this.current();
    if (!token) return null;

    switch (token.value.toUpperCase()) {
      case 'DEF':
      case 'SET':
        return this.parseVarDeclaration(token.value.toUpperCase() === 'DEF');
      case 'BUY':
      case 'SELL':
      case 'SELLSHORT':
        return this.parseTradeCommand(token.value.toUpperCase());
      case 'EXIT':
      case 'CLOSE':
        return this.parseExitCommand();
      case 'TRAILSTOP':
        return this.parseTrailStop();
      case 'IF':
        return this.parseIfStatement();
      case 'LOOP':
      case 'WHILE':
        return this.parseLoop(token.value.toUpperCase());
      case 'AI_QUERY':
        return this.parseAIQuery();
      case 'AI_GENERATE_SCRIPT':
        return this.parseAIGenerate();
      case 'ANALYZE_LOG':
        return this.parseAnalyzeLog();
      case 'RUN_ML':
        return this.parseRunML();
      case 'CLAW_WEB':
        return this.parseClawWeb();
      case 'CLAW_X':
        return this.parseClawX();
      case 'CLAW_PDF':
        return this.parseClawPdf();
      case 'CLAW_IMAGE':
        return this.parseClawImage();
      case 'CLAW_VIDEO':
        return this.parseClawVideo();
      case 'CLAW_IMAGE_VIEW':
        return this.parseClawImageView();
      case 'SPAWN_AGENT':
        return this.parseSpawnAgent();
      case 'CALL_SESSION':
        return this.parseCallSession();
      case 'MUTATE_CONFIG':
        return this.parseMutateConfig();
      case 'ALERT':
        return this.parseAlert();
      case 'STORE_VAR':
        return this.parseStoreVar();
      case 'LOAD_VAR':
        return this.parseLoadVar();
      case 'WAIT':
        return this.parseWait();
      case 'TRY':
        return this.parseTryCatch();
      case 'ERROR':
        return this.parseError();
      case 'INCLUDE':
        return this.parseInclude();
      case 'OPTIMIZE':
        return this.parseOptimize();
      case 'INDICATOR':
        return this.parseIndicator();
      default:
        if (token.type === TOKEN_TYPES.IDENTIFIER) {
          return this.parseAssignment();
        }
        throw new Error(`Unexpected token: ${token.value}`);
    }
  }

  // Example parse functions (abbreviating some for brevity; expand as needed)
  parseVarDeclaration(isDef) {
    this.eat(TOKEN_TYPES.IDENTIFIER); // DEF/SET
    const name = this.eat(TOKEN_TYPES.IDENTIFIER).value;
    this.eat(TOKEN_TYPES.OPERATOR); // =
    const value = this.parseExpression();
    if (isDef && this.variables.has(name)) throw new Error(`Var ${name} already defined`);
    this.variables.set(name, value);
    return { type: 'VarDecl', name, value, isDef };
  }

  parseTradeCommand(type) {
    this.eat(TOKEN_TYPES.IDENTIFIER); // BUY/SELL/etc.
    const size = this.current().type === TOKEN_TYPES.NUMBER ? this.eat(TOKEN_TYPES.NUMBER).value : 'config.size';
    this.imports.add('trade'); // Flag OpenClaw trade module
    if (this.current().value.toUpperCase() === 'AT') {
      this.eat(TOKEN_TYPES.IDENTIFIER); // AT
      const orderType = this.eat(TOKEN_TYPES.IDENTIFIER).value.toUpperCase(); // MARKET/LIMIT
      if (orderType === 'LIMIT') {
        this.eat(TOKEN_TYPES.PUNCTUATION); // (
        const price = this.parseExpression();
        this.eat(TOKEN_TYPES.PUNCTUATION); // )
      }
    }
    const condition = this.current().value.toUpperCase() === 'IF' ? this.parseCondition() : true;
    const stop = this.current().value.toUpperCase() === 'STOP' ? this.parseDistance('STOP') : 'config.stopDistance';
    const limit = this.current().value.toUpperCase() === 'LIMIT' ? this.parseDistance('LIMIT') : 'config.limitDistance';
    const reason = this.current().value.toUpperCase() === 'REASON' ? this.parseReason() : '"Auto"';
    return { type: 'Trade', command: type, size, condition, stop, limit, reason };
  }

  parseAIQuery() {
    this.eat(TOKEN_TYPES.IDENTIFIER); // AI_QUERY
    const prompt = this.eat(TOKEN_TYPES.STRING).value;
    const tool = this.current().value.toUpperCase() === 'TOOL' ? this.parseParam('TOOL') : null;
    const arg = this.current().value.toUpperCase() === 'ARG' ? this.parseParam('ARG') : null;
    this.imports.add('ai'); // OpenClaw AI module
    return { type: 'AIQuery', prompt, tool, arg };
  }

  // [Implement similar parse* for all commands: parseExitCommand, parseTrailStop, parseIfStatement, parseLoop, parseAIGenerate, parseAnalyzeLog, parseRunML, parseClawWeb, parseClawX, parseClawPdf, parseClawImage, parseClawVideo, parseClawImageView, parseSpawnAgent, parseCallSession, parseMutateConfig, parseAlert, parseStoreVar, parseLoadVar, parseWait, parseTryCatch, parseError, parseInclude, parseOptimize, parseIndicator, parseAssignment, parseExpression (for conditions/math), parseCondition, parseDistance, parseReason, parseParam]

  // Helper: parseExpression (recursive for math/ops)
  parseExpression() {
    let left = this.parsePrimary();
    while (this.current() && this.current().type === TOKEN_TYPES.OPERATOR) {
      const op = this.eat(TOKEN_TYPES.OPERATOR).value;
      const right = this.parsePrimary();
      left = { type: 'BinaryExpr', left, op, right };
    }
    return left;
  }

  parsePrimary() {
    const token = this.current();
    if (token.type === TOKEN_TYPES.NUMBER) return this.eat(TOKEN_TYPES.NUMBER);
    if (token.type === TOKEN_TYPES.STRING) return this.eat(TOKEN_TYPES.STRING);
    if (token.type === TOKEN_TYPES.IDENTIFIER) {
      const id = this.eat(TOKEN_TYPES.IDENTIFIER);
      if (this.current().value === '(') {
        this.eat(TOKEN_TYPES.PUNCTUATION); // (
        const args = [];
        while (this.current().value !== ')') {
          args.push(this.parseExpression());
          if (this.current().value === ',') this.eat(TOKEN_TYPES.PUNCTUATION);
        }
        this.eat(TOKEN_TYPES.PUNCTUATION); // )
        return { type: 'CallExpr', callee: id, args };
      }
      return id;
    }
    if (this.current().value === '(') {
      this.eat(TOKEN_TYPES.PUNCTUATION);
      const expr = this.parseExpression();
      this.eat(TOKEN_TYPES.PUNCTUATION);
      return expr;
    }
    throw new Error(`Unexpected primary: ${token.value}`);
  }

  // Code Generation: AST to JS string
  generateJS(ast, strategyName) {
    let js = `const BaseStrategy = require('./base-strategy.cjs');\n`;
    this.imports.forEach(imp => js += `const { ${imp} } = require('../openclaw-${imp}.cjs'); // Assume OpenClaw modules\n`);
    js += `\nclass ${strategyName}Strategy extends BaseStrategy {\n`;
    js += '  constructor(config) { super(config); }\n\n';
    js += '  evaluateEntry(ticks, context) {\n';
    ast.body.forEach(stmt => js += this.generateStmt(stmt, '    '));
    js += '    return null; // Default no signal\n  }\n\n';
    js += '  // [Similar for evaluateExit if exits in AST]\n';
    js += '  getRequiredBufferSize() { return /* calc from AST */; }\n';
    js += '  getConfigSchema() { return /* from vars */; }\n';
    js += '  static get STRATEGY_TYPE() { return "custom-${strategyName.toLowerCase()}"; }\n';
    js += '};\n\nmodule.exports = ${strategyName}Strategy;\n';
    return js;
  }

  generateStmt(stmt, indent = '') {
    switch (stmt.type) {
      case 'VarDecl':
        return `${indent}${stmt.isDef ? 'const' : 'let'} ${stmt.name} = ${this.generateExpr(stmt.value)};\n`;
      case 'Trade':
        return `${indent}if (${this.generateExpr(stmt.condition)}) {\n${indent}  return { signal: true, direction: "${stmt.command.replace('SELLSHORT', 'SELL')}", size: ${stmt.size}, stopDist: ${stmt.stop}, limitDist: ${stmt.limit}, reason: ${stmt.reason} };\n${indent}}\n`;
      case 'AIQuery':
        return `${indent}const ${stmt.var || '_result'} = await aiQuery("${stmt.prompt}", { tool: "${stmt.tool}", arg: "${stmt.arg}" });\n`;
      // [Implement generate* for all stmt types: similar conditional/return/await patterns, calling OpenClaw functions]
      default:
        return `${indent}// TODO: ${stmt.type}\n`;
    }
  }

  generateExpr(expr) {
    if (typeof expr === 'number' || typeof expr === 'string') return JSON.stringify(expr);
    if (expr.type === 'Identifier') return expr.value;
    if (expr.type === 'BinaryExpr') return `(${this.generateExpr(expr.left)} ${expr.op} ${this.generateExpr(expr.right)})`;
    if (expr.type === 'CallExpr') {
      const args = expr.args.map(a => this.generateExpr(a)).join(', ');
      if (expr.callee.value.toUpperCase() === 'INDICATOR') {
        this.imports.add('indicators');
        return `indicators.calc${expr.args[0].value}(${args.slice(2)})`;
      }
      // [Handle other calls like CLAW_WEB: `await clawWeb(${args})`]
      return `${expr.callee.value}(${args})`;
    }
    return '/* expr */';
  }
}

// Main parse + generate function
function parseAndGenerate(code, strategyName) {
  const tokens = lexer(code);
  const parser = new ClawScriptParser(tokens);
  const ast = parser.parse();
  return parser.generateJS(ast, strategyName);
}

module.exports = { parseAndGenerate, lexer, ClawScriptParser };
```
