### 1. Skill Integration (Calling OpenClaw Tools)
We'll add a new command: **`CLAW_TOOL`** — lets scripts call any OpenClaw tool (e.g., code_execution for ML, browse_page for web data, x_semantic_search for X posts). This is like a supercharged `AI_QUERY` but for all tools.

**Grammar**:
```text
DEF var = CLAW_TOOL "tool_name" WITH args [WAIT_TIMEOUT seconds]
```

- `tool_name`: string — e.g., "code_execution", "browse_page", "x_semantic_search"
- `args`: key-value pairs (e.g., `QUERY "search term" LIMIT 5`)
- `WAIT_TIMEOUT`: optional — for async tools

**Examples**:
```clawscript
// Run Python code for ML prediction
DEF prediction = CLAW_TOOL "code_execution" WITH CODE "import numpy; return numpy.mean(prices)"

 آشنا// Browse web and summarize
DEF summary = CLAW_TOOL "browse_page" WITH URL "https://news.com.au" INSTRUCTIONS "Extract rumors about AUD/USD"

ключа// Search X for rumors
DEF posts = CLAW_TOOL "x_semantic_search" WITH QUERY "AUD economy rumors" LIMIT 5 MIN_SCORE 0.2
IF posts.length > 0 THEN
  ALERT "Rumors found: " + posts[0].text TO "telegram"
ENDIF
```

**How It Works**:
- Parser generates JS calling your OpenClaw tool wrappers (e.g., `await tools.code_execution({code: '...'})`).
- Ties into full power: Scripts can now run bio/chem/physics sims (via code_execution libs like rdkit/biopython), fetch real-time data, or chain tools (e.g., web search → analyze with ML).

### 2. Channel Integration (SMS, Telegram, Voice Output)
Extend the existing `ALERT` command with more `TO` options. For voice, use a text-to-speech wrapper (e.g., via code_execution with a lib like gtts).

**Updated Grammar for ALERT**:
```text
ALERT "message" [LEVEL "info/warn/error"] [TO "channel" OPTIONS "key=value"]
```

- `channel`: "telegram", "sms", "voice", "email", "discord" (expand as needed)
- `OPTIONS`: channel-specific (e.g., `PHONE "+614..."` for SMS/voice)

**Examples**:
```clawscript
ALERT "Trade alert: Bought AUD at 0.65" LEVEL "info" TO "telegram" OPTIONS "chat_id=12345"

ALERT "Emergency drawdown!" LEVEL "error" TO "sms" OPTIONS "phone=+61412345678"

ALERT "Market summary: " + summary LEVEL "info" TO "voice" OPTIONS "phone=+61412345678 lang=en-AU"  // TTS call
```

**How It Works**:
- Generator calls a new `openclaw-channels.cjs` wrapper: e.g., `await channels.send("telegram", message, options)`.
- For Telegram/SMS: Use Twilio/Telegram API (add keys to env).
- For Voice: Use code_execution with gtts (generate MP3) + Twilio for call.
- Non-trading use: Scripts can notify you about anything (e.g., "X mention detected").

### 3. Rumor Scan (Specialized Command)
New command: **`RUMOR_SCAN`** — combines x_semantic_search + web_search + AI summary for rumor detection.

**Grammar**:
```text
DEF var = RUMOR_SCAN "topic" [SOURCES "x/web/both"] [LIMIT num] [FILTER "keyword"]
```

- `topic`: search query (e.g., "AUD crash rumors")
- `SOURCES`: "x" (X only), "web" (news/sites), "both" (default)
- Returns: array of {source, text, score, link}

**Example**:
```clawscript
DEF rumors = RUMOR_SCAN "Brisbane economy rumors" SOURCES "both" LIMIT 10 FILTER "negative"
IF rumors.length > 0 THEN
  SAY_TO_SESSION "ceo" "Rumors detected: " + rumors[0].text
  ALERT "Alert on rumors" TO "voice" OPTIONS "phone=+614..."
ENDIF
```

**How It Works**:
- Parser generates JS chaining tools: e.g., `await tools.x_semantic_search(...) + tools.web_search(...) + ai.summarize(results, "Detect rumors")`.

### Expanded Parser Code Snippets

Add these to `clawscript-parser.cjs`:

#### New Parse Methods

```javascript
parseClawTool() {
  this.eat(TOKEN_TYPES.KEYWORD); // CLAW_TOOL
  const toolName = this.parseExpression();
  this.eat(TOKEN_TYPES.KEYWORD); // WITH
  const args = {};
  while (this.current().type === TOKEN_TYPES.IDENTIFIER && this.peek().value === '=') {
    const key = this.eat(TOKEN_TYPES.IDENTIFIER).value.toUpperCase();
    this.eat(TOKEN_TYPES.OPERATOR); // =
    args[key] = this.parseExpression();
  }
  let timeout = 300;
  if (this.current().value.toUpperCase() === 'WAIT_TIMEOUT') {
    this.eat(TOKEN_TYPES.KEYWORD);
    timeout = this.parseExpression();
  }
  this.imports.add('tools');
  return { type: 'ClawTool', toolName, args, timeout };
}

parseAlert() {
  this.eat(TOKEN_TYPES.KEYWORD); // ALERT
  const message = this.parseExpression();
  let level = '"info"';
  if (this.current().value.toUpperCase() === 'LEVEL') {
    this.eat(TOKEN_TYPES.KEYWORD);
    level = this.parseExpression();
  }
  let channel = null;
  let options = {};
  if (this.current().value.toUpperCase() === 'TO') {
    this.eat(TOKEN_TYPES.KEYWORD);
    channel = this.parseExpression();
    if (this.current().value.toUpperCase() === 'OPTIONS') {
      this.eat(TOKEN_TYPES.KEYWORD);
      this.eat(TOKEN_TYPES.OPERATOR); // =
      options = this.parseKeyValuePairs(); // New helper for key=value
    }
  }
  this.imports.add('channels');
  return { type: 'Alert', message, level, channel, options };
}

parseRumorScan() {
  this.eat(TOKEN_TYPES.KEYWORD); // RUMOR_SCAN
  const topic = this.parseExpression();
  let sources = '"both"';
  if (this.current().value.toUpperCase() === 'SOURCES') {
    this.eat(TOKEN_TYPES.KEYWORD);
    sources = this.parseExpression();
  }
  let limit = 10;
  if (this.current().value.toUpperCase() === 'LIMIT') {
    this.eat(TOKEN_TYPES.KEYWORD);
    limit = this.parseExpression();
  }
  let filter = null;
  if (this.current().value.toUpperCase() === 'FILTER') {
    this.eat(TOKEN_TYPES.KEYWORD);
    filter = this.parseExpression();
  }
  this.imports.add('tools');
  this.imports.add('ai');
  return { type: 'RumorScan', topic, sources, limit, filter };
}

// Helper: parse key=value pairs (for OPTIONS, WITH args)
parseKeyValuePairs() {
  const pairs = {};
  while (this.current().type === TOKEN_TYPES.IDENTIFIER) {
    const key = this.eat(TOKEN_TYPES.IDENTIFIER).value;
    this.eat(TOKEN_TYPES.OPERATOR); // =
    pairs[key] = this.parseExpression();
    if (this.current().value === ',') this.eat(TOKEN_TYPES.PUNCTUATION);
  }
  return pairs;
}
```

case 'ClawTool':
  let jsTool = `${indent}const ${stmt.var || '_result'} = await tools.${stmt.toolName.toLowerCase()}(`;
  jsTool += `{ ${Object.entries(stmt.args).map(([k, v]) => `${k.toLowerCase()}: ${this.generateExpr(v)}`).join(', ')} }, `;

  jsTool += `);\n`;

case 'RumorScan':

  jsScan += `${indent}if (${this.generateExpr(stmt.sources)} === "web" || ${this.generateExpr(stmt.sources)} === "both") {\n`;
  jsScan += `${indent}  _rumors = _rumors.concat(await tools.web_search(${this.generateExpr(stmt.topic)}, { num_results: ${stmt.limit} }));\n`;
  jsScan += `${indent}}\n`;
  jsScan += `${indent}_rumors = await ai.summarize(_rumors, "Detect and score rumors");\n`;
  if (stmt.filter) {
    jsScan += `${indent}_rumors = _rumors.filter(r => r.text.includes(${this.generateExpr(stmt.filter)}));\n`;

  jsScan += `${indent}const ${stmt.var || '_result'} = _rumors;\n`;
